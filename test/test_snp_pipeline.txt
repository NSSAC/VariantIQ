fastq-dump --split-files ERR2704678

#H37Rv reference genome
wget -O Mycobacterium_tuberculosis_H37Rv.fasta "https://www.ncbi.nlm.nih.gov/sviewer/viewer.fcgi?id=NC_000962.3&db=nuccore&report=fasta&extrafeat=null&conwithfeat=on&hide-cdd=on"

#minos truth genome
wget -c https://figshare.com/ndownloader/files/30753355 -O - | tar -xz

#test snippy
#snippy --cpus 2 --outdir  test_snippy --prefix test --ref Mycobacterium_tuberculosis_H37Rv.fasta --R1 ERR2704678_1.fastq --R2 ERR2704678_2.fastq --unmapped --report --force
snippy --cpus 2 --ref Mycobacterium_tuberculosis_H37Rv.fasta --R1 ERR2704678_1.fastq --R2 ERR2704678_2.fastq --outdir snippy_output --basequal 20 --unmapped --report --force

#test SNP pipeline
bwa index Mycobacterium_tuberculosis_H37Rv.fasta
bwa mem Mycobacterium_tuberculosis_H37Rv.fasta ERR2704678_1.fastq ERR2704678_2.fastq -o ERR2704678_bwa-mem_aligned_reads.sam
samtools view -S -b ERR2704678_bwa-mem_aligned_reads.sam -o ERR2704678_bwa-mem_aligned_reads.bam
samtools sort ERR2704678_bwa-mem_aligned_reads.bam -o ERR2704678_bwa-mem_aligned_reads.sorted.bam
picard MarkDuplicates I=ERR2704678_bwa-mem_aligned_reads.sorted.bam O=dedup_ERR2704678_bwa-mem_aligned_reads.sorted.bam M=metrics.txt
samtools index dedup_ERR2704678_bwa-mem_aligned_reads.sorted.bam
bcftools mpileup -f Mycobacterium_tuberculosis_H37Rv.fasta dedup_ERR2704678_bwa-mem_aligned_reads.sorted.bam | bcftools call -mv -Ov -o ERR2704678_variants_bcftools.vcf
freebayes -f Mycobacterium_tuberculosis_H37Rv.fasta --ploidy 1 dedup_ERR2704678_bwa-mem_aligned_reads.sorted.bam -v ERR2704678_variants_freebayes.vcf

bcftools norm -f Mycobacterium_tuberculosis_H37Rv.fasta ERR2704678_variants_bcftools.vcf -o normalized_ERR2704678_variants_bcftools.vcf
bcftools norm -f Mycobacterium_tuberculosis_H37Rv.fasta ERR2704678_variants_freebayes.vcf -o normalized_ERR2704678_variants_freebayes.vcf

#vt normalize -r Mycobacterium_tuberculosis_H37Rv.fasta  ERR2704678_variants_freebayes.vcf -o vt_normalized_ERR2704678_variants_freebayes.vcf

minos adjudicate --reads ERR2704678_1.fastq --reads ERR2704678_2.fastq test_minos Mycobacterium_tuberculosis_H37Rv.fasta normalized_ERR2704678_variants_bcftools.vcf ERR2704678_variants_freebayes.vcf --force
varifier vcf_eval Minos.biorxiv.202109.data/Assemblies/mtb.N0054.fasta Mycobacterium_tuberculosis_H37Rv.fasta normalized_ERR2704678_variants_bcftools.vcf.gz test_varifier --force
